# Synthetic review fixture for the smoke test in docs/review/smoke-test/README.md.
# Not application code. Never imported, never executed. All names and data are invented.
# Fixture model (assumption of this fixture only, not a PayCore decision):
# amounts are signed integers in minor units with an explicit currency,
# and every tenant-scoped table has a tenant_id column.

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy import select
from sqlalchemy.orm import Session

from fixture_app.auth import AuthContext, require_api_key
from fixture_app.db import get_session
from fixture_app.models import Charge

router = APIRouter()


@router.get("/charges/{charge_id}")
def get_charge(
    charge_id: str,
    auth: AuthContext = Depends(require_api_key),
    session: Session = Depends(get_session),
):
    charge = session.execute(
        select(Charge).where(Charge.id == charge_id)
    ).scalar_one_or_none()
    if charge is None:
        raise HTTPException(status_code=404)
    return {
        "id": charge.id,
        "amount": charge.amount,
        "currency": charge.currency,
        "status": charge.status,
    }
